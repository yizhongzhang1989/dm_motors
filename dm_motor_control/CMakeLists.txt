cmake_minimum_required(VERSION 3.8)
project(dm_motor_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(dm_motor_interfaces REQUIRED)
add_library(dm_motor_core src/protocol.cpp src/transport.cpp src/controller.cpp)
target_include_directories(dm_motor_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
add_executable(motor_node src/motor_node.cpp)
target_link_libraries(motor_node dm_motor_core)
ament_target_dependencies(motor_node rclcpp std_srvs dm_motor_interfaces)
install(TARGETS dm_motor_core EXPORT dm_motor_core_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS motor_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(dm_motor_core_targets HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core dm_motor_core)
endif()
ament_package()
