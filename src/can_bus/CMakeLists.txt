cmake_minimum_required(VERSION 3.8)
project(can_bus)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(common REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Frame.msg" "msg/FrameFilter.msg" "msg/TxFrame.msg" "msg/Heartbeat.msg"
  "msg/Status.msg" "srv/RegisterClient.srv" "srv/Maintenance.srv"
  DEPENDENCIES std_msgs builtin_interfaces)
add_library(can_bus_transport src/transport.cpp src/router.cpp)
target_include_directories(can_bus_transport PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_executable(can_bus_node src/bus_node.cpp)
target_link_libraries(can_bus_node can_bus_transport)
ament_target_dependencies(can_bus_node rclcpp rcl_interfaces common)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(can_bus_node "${typesupport}")
install(TARGETS can_bus_transport EXPORT can_bus_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS can_bus_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/dashboard.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY web DESTINATION share/${PROJECT_NAME})
install(FILES README.md DASHBOARD.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/fastdds_transport.sh")
ament_export_targets(can_bus_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(rosidl_default_runtime)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_router test/test_router.cpp)
  target_link_libraries(test_router can_bus_transport)
  ament_add_pytest_test(test_latency test/test_latency.py TIMEOUT 45)
  ament_add_pytest_test(test_dds_environment test/test_dds_environment.py TIMEOUT 30)
  ament_add_pytest_test(test_dashboard test/test_dashboard.py TIMEOUT 30)
  ament_add_pytest_test(test_dashboard_integration test/test_dashboard_integration.py TIMEOUT 60)
  ament_add_pytest_test(test_dashboard_launch test/test_dashboard_launch.py TIMEOUT 30)
endif()
ament_package()
