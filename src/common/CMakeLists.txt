cmake_minimum_required(VERSION 3.8)
project(common)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_library(common_config src/config.cpp)
target_include_directories(common_config PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(common_config ament_index_cpp)
target_link_libraries(common_config yaml-cpp)
install(TARGETS common_config EXPORT common_config_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES ../../config/dm_motors.yaml DESTINATION share/${PROJECT_NAME}/config RENAME runtime.yaml)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(common_config_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(ament_index_cpp yaml-cpp)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_pytest_test(test_config test/test_config.py TIMEOUT 30)
  ament_add_gtest(test_config_cpp test/test_config.cpp)
  target_link_libraries(test_config_cpp common_config)
endif()
ament_package()
