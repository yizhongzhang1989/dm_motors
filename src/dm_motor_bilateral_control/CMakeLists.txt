cmake_minimum_required(VERSION 3.8)
project(dm_motor_bilateral_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(dm_motor_control REQUIRED)
find_package(dm_motor_interfaces REQUIRED)
find_package(common REQUIRED)
add_library(bilateral_core STATIC src/bilateral.cpp)
target_include_directories(bilateral_core PUBLIC include)
ament_target_dependencies(bilateral_core dm_motor_control)
add_executable(bilateral_control_node src/bilateral_control_node.cpp)
target_link_libraries(bilateral_control_node bilateral_core)
ament_target_dependencies(bilateral_control_node rclcpp rcl_interfaces std_srvs dm_motor_control dm_motor_interfaces common)
install(TARGETS bilateral_control_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/dashboard.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch web DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_bilateral test/test_bilateral.cpp)
  target_link_libraries(test_bilateral bilateral_core)
  ament_add_pytest_test(test_launch test/test_launch.py TIMEOUT 30)
  ament_add_pytest_test(test_dashboard test/test_dashboard.py TIMEOUT 45)
endif()
ament_package()
