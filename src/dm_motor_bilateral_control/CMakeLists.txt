cmake_minimum_required(VERSION 3.8)
project(dm_motor_bilateral_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(dm_motor_control REQUIRED)
find_package(dm_motor_interfaces REQUIRED)
find_package(common REQUIRED)
add_library(bilateral_core STATIC src/bilateral.cpp)
target_include_directories(bilateral_core PUBLIC include)
ament_target_dependencies(bilateral_core dm_motor_control)
add_executable(bilateral_control_node src/bilateral_control_node.cpp)
target_link_libraries(bilateral_control_node bilateral_core)
ament_target_dependencies(bilateral_control_node rclcpp rcl_interfaces std_srvs dm_motor_control dm_motor_interfaces common)
install(TARGETS bilateral_control_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/dashboard.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/auto_test.py DESTINATION lib/${PROJECT_NAME} RENAME auto_test)
install(PROGRAMS scripts/manual_test.py DESTINATION lib/${PROJECT_NAME} RENAME manual_test)
install(DIRECTORY launch web docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_bilateral test/test_bilateral.cpp)
  target_link_libraries(test_bilateral bilateral_core)
  ament_add_gtest(test_differential test/test_differential.cpp)
  target_link_libraries(test_differential bilateral_core)
  ament_add_pytest_test(test_differential_node test/test_differential_node.py TIMEOUT 45
    ENV "DM_BILATERAL_TEST_NODE=$<TARGET_FILE:bilateral_control_node>")
  add_executable(offline_bilateral_benchmark test/offline_benchmark/benchmark.cpp)
  target_link_libraries(offline_bilateral_benchmark bilateral_core dm_motor_control::joint_simulation)
  ament_target_dependencies(offline_bilateral_benchmark dm_motor_control dm_motor_interfaces common)
  ament_add_pytest_test(test_offline_benchmark test/offline_benchmark/test_regression.py TIMEOUT 90
    ENV "DM_OFFLINE_BENCHMARK=$<TARGET_FILE:offline_bilateral_benchmark>")
  add_executable(test_three_motor_coupling test/three_motor_rig/test_coupling.cpp)
  ament_add_test(test_three_motor_coupling COMMAND "$<TARGET_FILE:test_three_motor_coupling>"
    GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
  add_executable(three_motor_fixture test/three_motor_rig/fixture.cpp)
  target_link_libraries(three_motor_fixture bilateral_core dm_motor_control::joint_simulation)
  ament_target_dependencies(three_motor_fixture dm_motor_control dm_motor_interfaces common)
  ament_add_pytest_test(test_three_motor_rig test/three_motor_rig/test_three_motor_fixture.py
    TIMEOUT 45 ENV "DM_THREE_MOTOR_FIXTURE=$<TARGET_FILE:three_motor_fixture>"
    "OPENBLAS_NUM_THREADS=1" "OMP_NUM_THREADS=1" "PYTHONDONTWRITEBYTECODE=1")
  ament_add_gtest(test_velocity_bandwidth test/test_velocity_bandwidth.cpp)
  target_include_directories(test_velocity_bandwidth PRIVATE include)
  ament_add_pytest_test(test_synchronized_spring_node test/test_synchronized_spring_node.py TIMEOUT 60
    ENV "DM_BILATERAL_TEST_NODE=$<TARGET_FILE:bilateral_control_node>")
  ament_add_pytest_test(test_unilateral_node test/test_unilateral_node.py TIMEOUT 60
    ENV "DM_BILATERAL_TEST_NODE=$<TARGET_FILE:bilateral_control_node>")
  ament_add_gtest(test_stop_confirmation test/test_stop_confirmation.cpp)
  target_include_directories(test_stop_confirmation PRIVATE include)
  ament_target_dependencies(test_stop_confirmation rclcpp dm_motor_control dm_motor_interfaces)
  ament_add_pytest_test(test_stop_counter_reset_node test/test_stop_counter_reset_node.py TIMEOUT 60
    ENV "DM_BILATERAL_TEST_NODE=$<TARGET_FILE:bilateral_control_node>")
  ament_add_gtest(test_light_touch test/test_light_touch.cpp)
  target_link_libraries(test_light_touch bilateral_core dm_motor_control::joint_simulation)
  ament_add_pytest_test(test_launch test/test_launch.py TIMEOUT 30)
  ament_add_pytest_test(test_dashboard test/test_dashboard.py TIMEOUT 45)
  ament_add_pytest_test(test_auto_test_report test/test_auto_test_report.py TIMEOUT 30)
  ament_add_pytest_test(test_angle_report test/test_angle_report.py TIMEOUT 30)
  ament_add_pytest_test(test_stationarity test/test_stationarity.py TIMEOUT 30)
  ament_add_pytest_test(test_manual_test test/test_manual_test.py TIMEOUT 45)
  ament_add_pytest_test(test_manual_guard test/test_manual_guard.py TIMEOUT 30)
endif()
ament_package()
