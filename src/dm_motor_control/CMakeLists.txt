cmake_minimum_required(VERSION 3.8)
project(dm_motor_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(dm_motor_interfaces REQUIRED)
find_package(can_bus REQUIRED)
add_library(dm_motor_core src/protocol.cpp src/transport.cpp src/controller.cpp src/runtime.cpp)
target_include_directories(dm_motor_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(dm_motor_core can_bus)
add_library(dm_can_client src/can_transport.cpp)
target_include_directories(dm_can_client PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(dm_can_client rclcpp can_bus)
add_executable(motor_node src/motor_node.cpp)
target_link_libraries(motor_node dm_motor_core dm_can_client)
ament_target_dependencies(motor_node rclcpp rcl_interfaces rclcpp_action std_srvs sensor_msgs diagnostic_msgs dm_motor_interfaces)
add_executable(can_motor_simulator src/can_motor_simulator.cpp)
target_link_libraries(can_motor_simulator dm_motor_core)
ament_target_dependencies(can_motor_simulator rclcpp can_bus std_srvs)
install(TARGETS dm_motor_core EXPORT dm_motor_core_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS dm_can_client EXPORT dm_motor_core_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS motor_node can_motor_simulator DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/motor_client.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_targets(dm_motor_core_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(can_bus rclcpp)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core dm_motor_core)
  ament_add_gtest(test_runtime test/test_runtime.cpp)
  target_link_libraries(test_runtime dm_motor_core)
  ament_add_pytest_test(test_node_integration test/test_node_integration.py TIMEOUT 180)
  ament_add_pytest_test(test_split_processes test/test_split_processes.py TIMEOUT 120)
  ament_add_pytest_test(test_launch_composition test/test_launch_composition.py TIMEOUT 30)
endif()
ament_package()
