cmake_minimum_required(VERSION 3.8)
project(dm_motor_lead_follow)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(dm_motor_interfaces REQUIRED)
add_executable(lead_follow_node src/lead_follow_node.cpp)
target_include_directories(lead_follow_node PRIVATE include)
ament_target_dependencies(lead_follow_node rclcpp rcl_interfaces std_srvs dm_motor_interfaces)
install(TARGETS lead_follow_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_follow test/test_follow.cpp)
  target_include_directories(test_follow PRIVATE include)
  ament_add_pytest_test(test_integration test/test_integration.py TIMEOUT 120)
endif()
ament_package()
