cmake_minimum_required(VERSION 3.16)
project(rig_native_stream LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(dm_motor_interfaces REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/stream.cpp" RIG_SOURCE_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" RIG_CMAKE_SHA256)
# Reconfigure the embedded hashes if either checked-in build input changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS stream.cpp CMakeLists.txt)
configure_file(build_info.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/build_info.hpp" @ONLY)
add_executable(rig_stream stream.cpp)
target_include_directories(rig_stream PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(rig_stream PRIVATE cxx_std_17)
target_compile_options(rig_stream PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(rig_stream rclcpp dm_motor_interfaces)
